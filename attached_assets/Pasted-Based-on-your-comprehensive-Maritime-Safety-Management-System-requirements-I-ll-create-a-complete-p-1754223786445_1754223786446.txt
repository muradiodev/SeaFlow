## Database Structure (PostgreSQL)

### Core Tables

````sql
-- Users Table
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(100) UNIQUE NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    role ENUM('ship_captain', 'ship_crew', 'dpa', 'superintendent', 'operator') NOT NULL,
    ship_id UUID REFERENCES ships(id),
    is_active BOOLEAN DEFAULT true,
    last_login TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Ships Table
CREATE TABLE ships (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    imo_number VARCHAR(20) UNIQUE NOT NULL,
    flag_state VARCHAR(100),
    ship_type VARCHAR(100),
    gross_tonnage INTEGER,
    built_year INTEGER,
    classification_society VARCHAR(100),
    is_active BOOLEAN DEFAULT true,
    last_sync TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Certificates Table
CREATE TABLE certificates (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ship_id UUID REFERENCES ships(id),
    certificate_type VARCHAR(200) NOT NULL,
    certificate_name VARCHAR(300) NOT NULL,
    issuing_authority VARCHAR(200),
    certificate_number VARCHAR(100),
    issue_date DATE,
    expiry_date DATE,
    file_path VARCHAR(500),
    status ENUM('valid', 'expired', 'expiring_soon') DEFAULT 'valid',
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Forms Table (Template)
CREATE TABLE forms (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(300) NOT NULL,
    description TEXT,
    form_type ENUM('recurring', 'one_time') NOT NULL,
    recurrence_pattern JSONB, -- {type: 'weekly', interval: 1, days: [1,2,3]}
    is_active BOOLEAN DEFAULT true,
    requires_approval BOOLEAN DEFAULT false,
    approval_roles JSONB, -- ['dpa', 'superintendent']
    form_fields JSONB NOT NULL, -- Dynamic form structure
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Form Instances (Active periods for recurring forms)
CREATE TABLE form_instances (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    form_id UUID REFERENCES forms(id),
    instance_name VARCHAR(300),
    start_date TIMESTAMP NOT NULL,
    end_date TIMESTAMP NOT NULL,
    is_active BOOLEAN DEFAULT true,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Form Submissions
CREATE TABLE form_submissions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    form_id UUID REFERENCES forms(id),
    form_instance_id UUID REFERENCES form_instances(id),
    ship_id UUID REFERENCES ships(id),
    submitted_by UUID REFERENCES users(id),
    submission_data JSONB NOT NULL, -- Form field values
    attachments JSONB, -- File paths and metadata
    status ENUM('draft', 'submitted', 'approved', 'rejected') DEFAULT 'draft',
    digital_signature JSONB, -- Signature metadata
    approved_by UUID REFERENCES users(id),
    approved_at TIMESTAMP,
    submitted_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Documents Table (General document management)
CREATE TABLE documents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(300) NOT NULL,
    document_type VARCHAR(100),
    category VARCHAR(100),
    file_path VARCHAR(500),
    file_size BIGINT,
    mime_type VARCHAR(100),
    version VARCHAR(20) DEFAULT '1.0',
    ship_id UUID REFERENCES ships(id),
    is_public BOOLEAN DEFAULT false,
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Notifications Table
CREATE TABLE notifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    recipient_id UUID REFERENCES users(id),
    ship_id UUID REFERENCES ships(id),
    title VARCHAR(300) NOT NULL,
    message TEXT NOT NULL,
    notification_type ENUM('form_deadline', 'certificate_expiry', 'system_update', 'approval_request', 'general') NOT NULL,
    priority ENUM('low', 'medium', 'high', 'critical') DEFAULT 'medium',
    is_read BOOLEAN DEFAULT false,
    related_entity_type VARCHAR(100), -- 'form', 'certificate', etc.
    related_entity_id UUID,
    scheduled_at TIMESTAMP,
    sent_at TIMESTAMP,
    read_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Manuals Table
CREATE TABLE manuals (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(300) NOT NULL,
    content TEXT,
    category VARCHAR(100),
    tags JSONB, -- For search functionality
    version VARCHAR(20) DEFAULT '1.0',
    file_attachments JSONB, -- Photos, diagrams
    is_searchable BOOLEAN DEFAULT true,
    last_updated_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Audit Logs Table
CREATE TABLE audit_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    ship_id UUID REFERENCES ships(id),
    action VARCHAR(100) NOT NULL,
    entity_type VARCHAR(100) NOT NULL, -- 'form', 'certificate', 'user', etc.
    entity_id UUID,
    old_values JSONB,
    new_values JSONB,
    ip_address INET,
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- KPIs Table
CREATE TABLE kpis (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(200) NOT NULL,
    description TEXT,
    calculation_formula TEXT, -- How to calculate this KPI
    data_source VARCHAR(100), -- Which forms/tables to use
    category VARCHAR(100),
    unit VARCHAR(50),
    target_value DECIMAL(10,2),
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- KPI Values Table (Historical data)
CREATE TABLE kpi_values (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    kpi_id UUID REFERENCES kpis(id),
    ship_id UUID REFERENCES ships(id),
    period_start DATE,
    period_end DATE,
    value DECIMAL(10,2),
    calculated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Sync Status Table (For offline synchronization)
CREATE TABLE sync_status (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    ship_id UUID REFERENCES ships(id),
    entity_type VARCHAR(100) NOT NULL,
    entity_id UUID NOT NULL,
    action ENUM('create', 'update', 'delete') NOT NULL,
    sync_status ENUM('pending', 'synced', 'failed') DEFAULT 'pending',
    data JSONB, -- Entity data to sync
    error_message TEXT,
    attempts INTEGER DEFAULT 0,
    last_attempt TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- User Sessions Table (For authentication)
CREATE TABLE user_sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id),
    session_token VARCHAR(500) NOT NULL,
    device_info JSONB,
    ip_address INET,
    expires_at TIMESTAMP NOT NULL,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- File Uploads Table
CREATE TABLE file_uploads (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    original_name VARCHAR(300),
    stored_name VARCHAR(300),
    file_path VARCHAR(500),
    file_size BIGINT,
    mime_type VARCHAR(100),
    uploaded_by UUID REFERENCES users(id),
    entity_type VARCHAR(100), -- What this file belongs to
    entity_id UUID,
    is_temporary BOOLEAN DEFAULT false,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
````
